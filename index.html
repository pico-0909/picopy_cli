Picopy

یک فریم‌ورک سبک، قابل حمل و صفر-وابستگی برای ساخت رابط خط فرمان (CLI) با پایتون. با چند خط کد، برنامه‌ای حرفه‌ای، زیبا و تعاملی بسازید.

شروع کنید مرجع API
Python 3.9+ Zero-Dependency 6 Themes

چرا Picopy؟

ساخت یک رابط خط فرمان زیبا و تعاملی معمولاً نیازمند کتابخانه‌های متعدد و پیکربندی پیچیده است. هر کتابخانه وابستگی‌های خاص خودش را دارد، نسخه‌ی پایتون را محدود می‌کند و کاربر نهایی باید قبل از اجرا، چندین دستور pip install بزند.

Picopy این مشکل را از ریشه حل می‌کند. تمام کتابخانه‌های ثالث داخل پوشه‌ی پروژه قرار می‌گیرند تا کاربر نهایی نیازی به نصب چیزی نداشته باشد. یعنی کافیست پوشه‌ی پروژه را کپی کنید و برنامه‌تان همه‌جا کار کند، حتی در سیستم‌هایی که اینترنت ندارند یا دسترسی ادمین محدود است.

ویژگی‌های کلیدی

سبک و سریع

بدون وابستگی سنگین. راه‌اندازی فوری، اجرای بدون تأخیر، حافظه‌ی کم.

قابل حمل

همه‌چیز داخل یک پوشه. کپی کن و هر جا اجرا کن.

صفر-وابستگی

کاربر نهایی نیازی به pip، اینترنت یا نصب چیزی ندارد.

۶ تم آماده

dark, light, dracula, nord, gruvbox, mono. یا تم سفارشی خودت را بساز.

نصب

برای استفاده از Picopy، کافیست فایل WHL را دریافت کنید و با یک دستور نصبش کنید.

نصب با یک دستور

pip install picopy-1.0.0-py3-none-any.whl

pip به‌صورت خودکار وابستگی‌های Picopy (مثل questionary) را از PyPI دانلود و نصب می‌کند.

تأیید نصب

python -c "from picopy import App; print('OK')"

اگر OK دیدید، نصب موفق بوده است.

استفاده در کد

from picopy import App

اجرای برنامه با یک دستور

برای اینکه کاربر بتواند از هر پوشه‌ای دستور picopy را تایپ کند، توی pyproject.toml این رو اضافه کنید: 

[project.scripts]
picopy = "picopy:run"

و توی picopy.py یه تابع run() بسازید که برنامه‌ی اصلی رو اجرا کنه. بعد از نصب، از هر پوشه‌ای picopy کار می‌کند.

شروع سریع

بیایید با یک مثال ساده شروع کنیم.

قدم ۱: یک فایل بسازید

from picopy import App

app = App("MyTool")

@app.menu("Main")
def main(m):
    m.go("Say Hi", say_hi)
    m.quit()

def say_hi():
    app.print("Hello!")
    app.pause()

if __name__ == "__main__":
    app.run()

قدم ۲: اجرا کنید

python main.py

کلاس App

کلاس App مدیر کل برنامه است. تنظیمات، ثبت منوها و فرم‌ها، مدیریت ناوبری، چاپ پیام‌ها، و نمایش Spinner و Progress همه از این کلاس می‌آیند.

سازنده

App(title="App", version="", theme="dark", width=60, color_enabled=None)
پارامترنوعپیش‌فرضتوضیح
title رشته "App" عنوان برنامه که بالای هر صفحه نمایش داده می‌شود.
version رشته "" نسخه‌ی برنامه که زیر عنوان نمایش داده می‌شود.
theme رشته یا دیکشنری "dark" تم رنگی برنامه. می‌تواند اسم تم آماده باشد یا دیکشنری سفارشی.
width عدد صحیح 60 عرض خط جداکننده‌ی بالای صفحه.
color_enabled بولی یا None None فعال یا غیرفعال کردن رنگ‌ها. اگر None باشد، خودکار تشخیص داده می‌شود. 
app = App("MyTool", version="1.0.0", theme="dracula", width=70)

متد menu

@app.menu(name, title=None, icon="", color=None)

یک منو ثبت می‌کند.

پارامترنوعپیش‌فرضتوضیح
nameرشتهاجباریشناسه‌ی منو.
titleرشتهNoneعنوان نمایشی. اگر None باشد، از name استفاده می‌شود.
iconرشته""آیکون متنی.
colorرشتهNoneرنگ منو.
 
@app.menu("Main", title="منوی اصلی", icon="H", color="cyan")
def main(m):
    m.quit()

یک فرم ثبت می‌کند.

پارامترنوعپیش‌فرضتوضیح

titleرشتهNoneعنوان فرم.
iconرشته""آیکون فرم.
colorرشتهNoneرنگ فرم.
 
@app.form("Settings", title="تنظیمات", icon="S", color="yellow")
def settings(p):
    p.text("Username")

colorرشتهNoneرنگ متن.

متد ask

خروجی: رشته یا None.

تم برنامه را در حین اجرا تغییر می‌دهد.

کلاس Menu

کلاس Menu یک صفحه‌ی تعاملی است که چند دکمه نمایش می‌دهد.

متد go

m.go(label, target, icon="", color=None)

یک دکمه می‌سازد که به صفحه یا منوی دیگر می‌رود.

پارامترنوعپیش‌فرضتوضیح
labelرشتهاجباریمتن دکمه.
targetرشته یا تابعاجباریشناسه‌ی صفحه یا تابع.
iconرشته""آیکون دکمه.
colorرشتهNoneرنگ دکمه.
 
m.go("Settings", "settings")
m.go("Say Hi", say_hi)
m.go("Profile", "profile", icon="P", color="green")

متد do

m.do(label, function, icon="", color=None)

یک دکمه می‌سازد که یک تابع را اجرا می‌کند.

پارامترنوعپیش‌فرضتوضیح
labelرشتهاجباریمتن دکمه.
functionتابعاجباریتابعی که اجرا می‌شود.
iconرشته""آیکون دکمه.
colorرشتهNoneرنگ دکمه.
 
m.do("Say Hi", say_hi)
m.do("Exit", lambda: setattr(app, "running", False))

متد line

m.line()

یک خط جداکننده بین دکمه‌ها می‌سازد.

متد back

m.back(label=None)

یک دکمه‌ی برگشت می‌سازد.

پارامترنوعپیش‌فرضتوضیح

متد quit

m.quit(label="X Exit", message="Goodbye!")

یک دکمه‌ی خروج می‌سازد.

پارامترنوعپیش‌فرضتوضیح
labelرشته"X Exit"متن دکمه.
messageرشته"Goodbye!"پیامی که بعد از خروج چاپ می‌شود.

m.with_color(color)

متد title_of

عنوان منو را تغییر می‌دهد.

مثال ۱: Task Manager

from picopy import App
import json
from pathlib import Path

app = App("Tasks", theme="dracula")
DB = Path("tasks.json")

def load_tasks():
    return json.loads(DB.read_text()) if DB.exists() else []

def save_tasks(tasks):
    DB.write_text(json.dumps(tasks, indent=2))

@app.menu("Main", icon="T")
def main(m):
    m.do("List", list_tasks)
    m.go("Add Task", "add")
    m.line()
    m.quit()

def list_tasks():
    tasks = load_tasks()
    if not tasks:
        app.warn("No tasks yet!")
    else:
        for i, t in enumerate(tasks, 1):
            app.print(f"  {i}. {t['text']}")
    app.pause()

@app.form("Add", icon="+")
def add(p):
    p.text("Task")
    p.choose("Priority", ["Low", "Medium", "High"])

    @p.save
    def save(v):
        tasks = load_tasks()
        tasks.append({"text": v["task"]})
        save_tasks(tasks)
        app.ok("Task added!")
        app.pause()

if __name__ == "__main__":
    app.run()

مثال ۲: System Info

from picopy import App
import platform

app = App("System", theme="nord")

@app.menu("Main")
def main(m):
    m.do("Overview", overview)
    m.do("Disk", disk)
    m.line()
    m.quit()

def overview():
    app.print(f"OS: [cyan]{platform.system()}[/cyan]")
    app.print(f"Python: [cyan]{platform.python_version()}[/cyan]")
    app.pause()

def disk():
    with app.spinner("Checking disk..."):
        import time
        time.sleep(1)
    app.ok("Done!")
    app.pause()

if __name__ == "__main__":
    app.run()
